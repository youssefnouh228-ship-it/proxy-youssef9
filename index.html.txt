<script>
    // =================== خوادم VPN ===================
    const servers = [
        { name: "خادم أمريكا", flag: "🇺🇸", location: "نيويورك, USA", proxyUrl: "https://cors-anywhere.herokuapp.com/", ping: "45ms", active: false },
        { name: "خادم أوروبا", flag: "🇪🇺", location: "فرانكفورت, ألمانيا", proxyUrl: "https://corsproxy.io/", ping: "78ms", active: false },
        { name: "خادم آسيا", flag: "🇸🇬", location: "سنغافورة", proxyUrl: "https://api.allorigins.win/get?url=", ping: "112ms", active: false },
        { name: "خادم عالمي", flag: "🌍", location: "سحابة عالمية", proxyUrl: "https://cors-proxy.htmldriven.com/?url=", ping: "95ms", active: false }
    ];

    let currentServer = null;
    let currentProxyUrl = null;

    // عرض الخوادم
    function displayServers() {
        const container = document.getElementById('serversGrid');
        container.innerHTML = servers.map((server, index) => `
            <div class="server-card ${server.active ? 'active' : ''}">
                <div class="server-info">
                    <span class="server-flag">${server.flag}</span>
                    <div>
                        <h4>${server.name}</h4>
                        <p>${server.location}</p>
                    </div>
                </div>
                <div class="server-ping">⚡ ${server.ping}</div>
                <button class="connect-btn" onclick="connectToServer(${index})">
                    ${server.active ? '✅ متصل' : '🔌 اتصل'}
                </button>
            </div>
        `).join('');
    }

    // الاتصال بالخادم
    function connectToServer(index) {
        servers.forEach(s => s.active = false);
        servers[index].active = true;
        currentServer = servers[index];
        currentProxyUrl = currentServer.proxyUrl;
        
        displayServers();
        
        const statusLed = document.getElementById('statusLed');
        const statusText = document.getElementById('statusText');
        
        statusLed.classList.add('connected');
        statusText.innerHTML = `🟢 متصل على ${currentServer.name} - تم إخفاء الـ IP`;
        
        localStorage.setItem('activeServer', JSON.stringify({ name: currentServer.name, index: index }));
    }

    // =================== البحث - يفتح صفحة جديدة ===================
    function performSearch() {
        const query = document.getElementById('searchInput').value.trim();
        if (!query) {
            alert('الرجاء إدخال نص للبحث');
            return;
        }
        
        // فتح بحث جوجل في صفحة جديدة
        const searchUrl = 'https://www.google.com/search?q=' + encodeURIComponent(query);
        window.open(searchUrl, '_blank');
    }

    // =================== التصفح عبر الوكيل ===================
    function browseViaProxy() {
        let url = document.getElementById('urlInput').value.trim();
        const iframe = document.getElementById('browserFrame');
        const statusText = document.getElementById('statusText');
        
        if (!url) {
            alert('الرجاء إدخال رابط الموقع');
            return;
        }
        
        if (!currentProxyUrl) {
            alert('⚠️ الرجاء اختيار خادم VPN أولاً من القائمة أعلاه');
            return;
        }
        
        if (!url.startsWith('http://') && !url.startsWith('https://')) {
            url = 'https://' + url;
        }
        
        statusText.innerHTML = `🔄 جاري التصفح عبر ${currentServer.name}...`;
        
        if (currentProxyUrl.includes('allorigins')) {
            const proxyUrl = currentProxyUrl + encodeURIComponent(url);
            fetch(proxyUrl)
                .then(response => response.json())
                .then(data => {
                    const doc = iframe.contentWindow.document;
                    doc.open();
                    doc.write(data.contents || '<html><body><h3>لا يمكن عرض المحتوى</h3></body></html>');
                    doc.close();
                    statusText.innerHTML = `🟢 متصل على ${currentServer.name} - يتصفح: ${url}`;
                })
                .catch(() => {
                    statusText.innerHTML = `❌ فشل التحميل - جرب خادماً آخر`;
                });
        } else {
            const proxyUrl = currentProxyUrl + encodeURIComponent(url);
            iframe.src = proxyUrl;
            statusText.innerHTML = `🟢 متصل على ${currentServer.name} - يتصفح: ${url}`;
        }
        
        localStorage.setItem('lastUrl', url);
    }

    // تحميل الخادم المحفوظ
    function loadSavedServer() {
        const saved = localStorage.getItem('activeServer');
        if (saved) {
            try {
                const data = JSON.parse(saved);
                const serverIndex = data.index;
                if (serverIndex !== undefined && servers[serverIndex]) {
                    connectToServer(serverIndex);
                }
            } catch(e) {}
        }
        
        const lastUrl = localStorage.getItem('lastUrl');
        if (lastUrl) {
            document.getElementById('urlInput').value = lastUrl;
        }
    }

    // ربط الأحداث
    document.getElementById('searchBtn').addEventListener('click', performSearch);
    document.getElementById('searchInput').addEventListener('keypress', (e) => {
        if (e.key === 'Enter') performSearch();
    });
    document.getElementById('urlInput').addEventListener('keypress', (e) => {
        if (e.key === 'Enter') browseViaProxy();
    });

    // تشغيل الموقع
    displayServers();
    loadSavedServer();
    
    if (!currentProxyUrl && servers.length > 0) {
        connectToServer(0);
    }
</script>
